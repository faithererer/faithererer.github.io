# 确保 FFmpeg 和 FFprobe 在系统 PATH 中
if (-not (Get-Command ffmpeg -ErrorAction SilentlyContinue)) {
    Write-Error "FFmpeg 未安装或未添加到系统 PATH。请安装 FFmpeg 并重试。"
    exit
}

# 定义匹配的文件模式
$pattern = "小芷糕_2025-01-01_22-15-20_00*.ts"

# 获取所有匹配的 .ts 文件
$files = Get-ChildItem -Filter $pattern

if ($files.Count -eq 0) {
    Write-Output "未找到匹配的文件：$pattern"
    exit
}

foreach ($file in $files) {
    Write-Output "正在处理文件：$($file.Name)..."

    # 使用 ffprobe 提取所有帧的宽度
    ffprobe -v error -select_streams v:0 -show_entries frame=width -of csv=p=0 "$($file.FullName)" > widths.txt

    if (-Not (Test-Path widths.txt)) {
        Write-Warning "无法提取宽度信息，跳过文件：$($file.Name)"
        continue
    }

    # 读取所有宽度值并转换为整数
    $widths = Get-Content widths.txt | ForEach-Object { [int]$_ }

    if ($widths.Count -eq 0) {
        Write-Warning "未找到帧宽度信息，跳过文件：$($file.Name)"
        continue
    }

    # 找到最小宽度
    $minWidth = ($widths | Measure-Object -Minimum).Minimum

    # 计算目标高度以符合9:16比例，并确保为偶数
    $targetHeight = [math]::Ceiling(($minWidth * 16) / 9 / 2) * 2

    Write-Output "最小宽度：$minWidth 像素，目标高度：$targetHeight 像素"

    # 定义输出文件名
    $outputFile = "竖屏e_$($file.BaseName).ts"

    # 运行 FFmpeg 命令以调整视频尺寸和添加填充
    ffmpeg -i "$($file.FullName)" -vf "scale=$minWidth:$targetHeight:force_original_aspect_ratio=decrease, pad=$minWidth:$targetHeight:(ow-iw)/2:(oh-ih)/2:color=black" -c:v libx264 -c:a aac -shortest "$outputFile"

    # 检查输出文件是否生成
    if (Test-Path $outputFile) {
        Write-Output "成功生成：$outputFile"
    }
    else {
        Write-Warning "生成输出文件失败：$outputFile"
    }

    # 清理临时文件
    Remove-Item widths.txt -ErrorAction SilentlyContinue
}

Write-Output "所有文件已处理完毕。"
